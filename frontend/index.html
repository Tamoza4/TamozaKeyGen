client_integration.py 
from client_integration import LicenseValidator

# Initialize the validator
validator = LicenseValidator(
    base_url="https://key.tamoza.xyz",
    app_name="MyApp"
)

# Validate a license key
result = validator.validate("XXXX-YYYY-ZZZZ-WWWW")

if result.granted:
    print(f"Welcome, {result.owner_name}!")
    print(f"Class:  {result.key_class}")
else:
    print(f"Denied: {result.message}")
    exit(1)
